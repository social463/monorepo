```ts
import type { User } from "@prisma/client";
import { prisma } from "../lib/prisma";
import { voteInclude, type VoteWithRelations } from "./voting-service";

export interface CategoryCount {
  categorySlug: string;
  categoryName: string;
  count: number;
}

export interface UserProfile {
  user: User;
  totalVotesReceived: number;
  monthsRecognized: number;
  categoryBreakdown: CategoryCount[];
}

export async function getUserProfile(
  userId: string,
): Promise<UserProfile | null> {
  const user = await prisma.user.findUnique({ where: { id: userId } });
  if (!user) return null;

  const totalVotesReceived = await prisma.vote.count({
    where: { votedId: userId },
  });

  const distinctMonths = await prisma.vote.findMany({
    where: { votedId: userId },
    distinct: ["periodId"],
    select: { periodId: true },
  });

  const grouped = await prisma.vote.groupBy({
    by: ["categoryId"],
    where: { votedId: userId },
    _count: { _all: true },
  });
  const categories = await prisma.category.findMany({
    where: { id: { in: grouped.map((g) => g.categoryId) } },
  });
  const categoryById = new Map(
    categories.map((category) => [category.id, category]),
  );
  const categoryBreakdown: CategoryCount[] = grouped
    .map((g) => {
      const category = categoryById.get(g.categoryId);
      return {
        categorySlug: category?.slug ?? g.categoryId,
        categoryName: category?.name ?? g.categoryId,
        count: g._count._all,
      };
    })
    .sort(
      (a, b) =>
        b.count - a.count || a.categoryName.localeCompare(b.categoryName),
    );

  return {
    user,
    totalVotesReceived,
    monthsRecognized: distinctMonths.length,
    categoryBreakdown,
  };
}

interface VoteFilters {
  month?: string;
  categorySlug?: string;
}

export function listVotesReceived(
  userId: string,
  filters: VoteFilters,
): Promise<VoteWithRelations[]> {
  return prisma.vote.findMany({
    where: {
      votedId: userId,
      ...(filters.month ? { period: { monthRef: filters.month } } : {}),
      ...(filters.categorySlug
        ? { category: { slug: filters.categorySlug } }
        : {}),
    },
    include: voteInclude,
    orderBy: { createdAt: "desc" },
  });
}
```

```ts
import type { FastifyInstance } from "fastify";
import { prisma } from "../lib/prisma";
import { getUserProfile, listVotesReceived } from "../services/profile-service";
import { listBadgesForUser } from "../services/badge-service";
import { toAwardedBadgeDTO, toPublicUser, toVoteDTO } from "../lib/serialize";

export async function profileRoutes(app: FastifyInstance) {
  app.get(
    "/users/:id",
    { onRequest: [app.authenticate] },
    async (request, reply) => {
      const { id } = request.params as { id: string };
      const user = await prisma.user.findUnique({ where: { id } });
      if (!user) {
        return reply.code(404).send({ message: "Usuário não encontrado" });
      }
      return reply.send({ user: toPublicUser(user) });
    },
  );

  app.get(
    "/users/:id/profile",
    { onRequest: [app.authenticate] },
    async (request, reply) => {
      const { id } = request.params as { id: string };
      const profile = await getUserProfile(id);
      if (!profile) {
        return reply.code(404).send({ message: "Usuário não encontrado" });
      }
      const badges = await listBadgesForUser(id);
      return reply.send({
        user: toPublicUser(profile.user),
        stats: {
          totalVotesReceived: profile.totalVotesReceived,
          monthsRecognized: profile.monthsRecognized,
        },
        categoryBreakdown: profile.categoryBreakdown,
        badges: badges.map(toAwardedBadgeDTO),
      });
    },
  );

  app.get(
    "/users/:id/votes",
    { onRequest: [app.authenticate] },
    async (request, reply) => {
      const { id } = request.params as { id: string };
      const { month, categorySlug } = request.query as {
        month?: string;
        categorySlug?: string;
      };
      const votes = await listVotesReceived(id, { month, categorySlug });
      return reply.send({ votes: votes.map(toVoteDTO) });
    },
  );
}
```

```tsx
        `/users/${id}/votes${categorySlug ? `?categorySlug=${categorySlug}` : ""}`,
      ),
    enabled: Boolean(id),
  });

  if (profileQuery.isLoading) {
    return <p className="p-8 font-mono text-secondary">Carregando perfil…</p>;
  }
  if (profileQuery.isError || !profileQuery.data) {
    return <p className="p-8 text-red-400">Perfil não encontrado.</p>;
  }

  const profile = profileQuery.data;

  return (
    <main className="min-h-screen p-8 flex flex-col gap-6">
      <header className="flex items-start justify-between">
        <div>
          <h1 className="text-3xl font-extrabold tracking-tight">
            {profile.user.name}
          </h1>
          <p className="font-mono text-xs text-secondary">
            {profile.user.position ?? "—"} · {profile.user.squad ?? "—"}
          </p>
        </div>
        <Link to="/" className="text-sm text-slate-400 hover:text-slate-100">
          ← Voltar
        </Link>
      </header>

      <section className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
        <div className="bg-surface rounded-lg p-4">
          <p className="font-mono text-xs uppercase tracking-widest text-secondary">
            Votos recebidos
          </p>
          <p className="text-2xl font-extrabold">
            {profile.stats.totalVotesReceived}
          </p>
        </div>
        <div className="bg-surface rounded-lg p-4">
          <p className="font-mono text-xs uppercase tracking-widest text-secondary">
            Meses reconhecido
          </p>
          <p className="text-2xl font-extrabold">
            {profile.stats.monthsRecognized}
          </p>
        </div>
      </section>

      <section>
        <h2 className="font-mono text-xs uppercase tracking-widest text-secondary mb-3">
          Selos conquistados
        </h2>
        {profile.badges.length === 0 ? (
          <p className="text-slate-400">Nenhum selo ainda.</p>
        ) : (
          <ul className="flex flex-wrap gap-2">
            {profile.badges.map((awarded) => (
              <li
                key={awarded.id}
                className="bg-surface border border-primary rounded-lg px-3 py-2 text-sm"
              >
                {awarded.badge.name}
              </li>
            ))}
          </ul>
        )}
      </section>

      <section>
        <h2 className="font-mono text-xs uppercase tracking-widest text-secondary mb-3">
          Categorias mais reconhecidas
        </h2>
        {profile.categoryBreakdown.length === 0 ? (
          <p className="text-slate-400">Ainda sem reconhecimentos.</p>
        ) : (
          <ul className="flex flex-col gap-1">
            {profile.categoryBreakdown.map((item) => (
              <li
                key={item.categorySlug}
                className="flex justify-between bg-surface rounded px-3 py-2"
              >
                <span>{item.categoryName}</span>
                <span className="font-mono text-secondary">{item.count}</span>
              </li>
            ))}
          </ul>
        )}
      </section>

      <section>
        <div className="flex items-center justify-between mb-3">
          <h2 className="font-mono text-xs uppercase tracking-widest text-secondary">
            Reconhecimentos recebidos
          </h2>
          <select
            value={categorySlug}
            onChange={(event) => setCategorySlug(event.target.value)}
            aria-label="Filtrar por categoria"
            className="rounded bg-bg border border-slate-700 px-2 py-1 text-sm text-slate-100"
          >
            <option value="">Todas as categorias</option>
            {profile.categoryBreakdown.map((item) => (
              <option key={item.categorySlug} value={item.categorySlug}>
                {item.categoryName}
              </option>
            ))}
          </select>
        </div>
        {votesQuery.data && votesQuery.data.votes.length === 0 && (
          <p className="text-slate-400">Nenhum reconhecimento nesse filtro.</p>
        )}
        <ul className="flex flex-col gap-2">
          {votesQuery.data?.votes.map((vote) => (
            <li key={vote.id} className="bg-surface rounded-lg p-4">
              <p className="font-semibold">
                {vote.category.name}{" "}
                <span className="font-mono text-xs text-secondary">
                  · {vote.monthRef}
                </span>
              </p>
              <p className="text-sm text-slate-300">{vote.justification}</p>
              <p className="font-mono text-[11px] text-slate-500 mt-1">
                por {vote.voter.name}
              </p>
            </li>
          ))}
        </ul>
      </section>
    </main>
  );
}
```
