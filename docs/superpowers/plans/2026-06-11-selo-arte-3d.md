# Imagem de centro do selo via ilustrações 3D (3dicons.co) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Permitir que o admin escolha, numa grade clicável, uma ilustração 3D para o centro do selo, e redesenhar o medalhão (`BadgeEmblem`) com um visual mais ousado.

**Architecture:** O centro do selo deixa de ser um motivo geométrico e passa a ser uma ilustração 3D (assets estáticos CC0 do 3dicons.co em `public/badge-art/`). A identidade da ilustração é guardada no campo `iconKey` já existente (sem migração, sem mudança no DTO/API). Um manifesto em `lib/badge-art.ts` mapeia chave → asset + cor de acento. O `BadgeEmblem` renderiza a ilustração (com fallback para o motivo geométrico) e o `BadgeArtPicker` mostra a grade no formulário do admin.

**Tech Stack:** React 18 + TypeScript, Vite, Vitest + Testing Library, Tailwind. Monorepo pnpm (`@legends/web`, `@legends/api`).

// versionados em apps/web/public/badge-art/<key>.png.
// `color` é a cor de acento do medalhão escolhida para combinar com a arte.

export interface BadgeArt {
  key: string
  label: string
  src: string
  color: string

function art(key: string, label: string, color: string): BadgeArt {

}

  art('tools', 'Ferramentas', '#94a3b8'),
  art('globe', 'Globo', '#38bdf8'),
]

export const DEFAULT_ART_KEY = 'trophy'

const ART_BY_KEY = new Map(BADGE_ART.map((a) => [a.key, a]))

export function badgeArt(key: string): BadgeArt | undefined {
  return ART_BY_KEY.get(key)
}

const KIND_FALLBACK_COLOR: Record<string, string> = {
  IMPACT: '#facc15',
  RECURRENCE: '#2dd4bf',
  CATEGORY: '#52fba2',
  HIGHLIGHT: '#fb923c',
}

/** Cor do medalhão: cor da arte escolhida, ou cor por kind, ou verde padrão. */
export function badgeAccentColor(iconKey: string, kind: string): string {
  return badgeArt(iconKey)?.color ?? KIND_FALLBACK_COLOR[kind] ?? '#52fba2'
}

```tsx
import { useId } from 'react'
import type { BadgeDTO } from '@legends/shared'
import { badgeArt, badgeAccentColor } from '../lib/badge-art'

/** Pontos de uma estrela de 5 pontas centrada em (0,0). */
function starPoints(outer: number, inner: number): string {
  const pts: string[] = []
  for (let i = 0; i < 5; i++) {
    const ao = ((-90 + i * 72) * Math.PI) / 180
    pts.push(`${(outer * Math.cos(ao)).toFixed(2)},${(outer * Math.sin(ao)).toFixed(2)}`)
    const ai = ((-90 + i * 72 + 36) * Math.PI) / 180
    pts.push(`${(inner * Math.cos(ai)).toFixed(2)},${(inner * Math.sin(ai)).toFixed(2)}`)
  }
  return pts.join(' ')
}

/** Motivo geométrico de fallback (selos cuja iconKey não está no catálogo). */
function Motif({ kind }: { kind: BadgeDTO['kind'] }) {
  if (kind === 'IMPACT') return <polygon points={starPoints(17, 7)} />
  if (kind === 'HIGHLIGHT') return <polygon points={starPoints(18, 7.5)} />
  if (kind === 'RECURRENCE') {
    return (
      <>
        <circle r="17" fill="none" stroke="#ffffff" strokeOpacity="0.9" strokeWidth="2.4" />
        <polygon points={starPoints(10.5, 4.4)} />
      </>
    )
  }
  // CATEGORY → escudo
  return <path d="M -13,-14 L 13,-14 L 13,1 Q 13,11 0,16 Q -13,11 -13,1 Z" />
}

interface BadgeEmblemProps {
  badge: Pick<BadgeDTO, 'iconKey' | 'kind' | 'name'>
  /** Lado do emblema em px. */
  size?: number
}

const RAYS = Array.from({ length: 12 }, (_, i) => i * 30)
const RIVETS = Array.from({ length: 12 }, (_, i) => (i * 30 + 15) * (Math.PI / 180))
// Faíscas decorativas: posição angular (graus), raio e tamanho.
const SPARKLES = [
  { a: -62, r: 46, s: 3.2 },
  { a: 26, r: 47, s: 2.4 },
  { a: 152, r: 45, s: 2.8 },
  { a: 212, r: 46, s: 2.0 },
]

/** Estrela de 4 pontas (faísca) centrada em (0,0). */
function sparkPoints(s: number): string {
  const c = s * 0.28
  return `0,${-s} ${c},${-c} ${s},0 ${c},${c} 0,${s} ${-c},${c} ${-s},0 ${-c},${-c}`
}

export function BadgeEmblem({ badge, size = 64 }: BadgeEmblemProps) {
  const art = badgeArt(badge.iconKey)
  const color = badgeAccentColor(badge.iconKey, badge.kind)
  const uid = useId().replace(/:/g, '')

  return (
    <span
      className="badge-emblem inline-block shrink-0"
      style={{ width: size, height: size, ['--emblem-color' as string]: color }}
      title={badge.name}
      aria-label={badge.name}
      role="img"
    >
      <svg viewBox="0 0 100 100" width={size} height={size}>
        <defs>
          <radialGradient id={`dome-${uid}`} cx="38%" cy="30%" r="75%">
            <stop offset="0%" stopColor="#ffffff" stopOpacity="0.6" />
            <stop offset="32%" stopColor={color} />
            <stop offset="78%" stopColor={color} />
            <stop offset="100%" stopColor="#000000" stopOpacity="0.45" />
          </radialGradient>
          <linearGradient id={`ring-${uid}`} x1="0" y1="0" x2="1" y2="1">
            <stop offset="0%" stopColor="#ffffff" stopOpacity="0.55" />
            <stop offset="45%" stopColor={color} />
            <stop offset="100%" stopColor="#000000" stopOpacity="0.5" />
          </linearGradient>
          <linearGradient id={`ribbon-${uid}`} x1="0" y1="0" x2="0" y2="1">
            <stop offset="0%" stopColor={color} />
            <stop offset="100%" stopColor="#000000" stopOpacity="0.55" />
          </linearGradient>
          <linearGradient id={`shine-${uid}`} x1="0" y1="0" x2="1" y2="0">
            <stop offset="0%" stopColor="#ffffff" stopOpacity="0" />
            <stop offset="50%" stopColor="#ffffff" stopOpacity="0.55" />
            <stop offset="100%" stopColor="#ffffff" stopOpacity="0" />
          </linearGradient>
          <filter id={`emb-${uid}`} x="-20%" y="-20%" width="140%" height="140%">
            <feDropShadow dx="0" dy="1" stdDeviation="0.6" floodColor="#000000" floodOpacity="0.45" />
          </filter>
          <clipPath id={`dome-clip-${uid}`}>
            <circle cx="0" cy="0" r="33" />
          </clipPath>
        </defs>

        <g transform="translate(50,50)">
          {/* fita (ribbon) decorativa na base */}
          <g>
            <polygon points="-16,28 -7,28 -10,48 -20,42" fill={`url(#ribbon-${uid})`} />
            <polygon points="16,28 7,28 10,48 20,42" fill={`url(#ribbon-${uid})`} />
          </g>

          {/* raios de explosão */}
          <g opacity="0.18">
            {RAYS.map((deg) => (
              <polygon key={deg} points="0,-49 -2.2,-30 2.2,-30" fill={color} transform={`rotate(${deg})`} />
            ))}
          </g>

          {/* anel metálico */}
          <circle r="44" fill={`url(#ring-${uid})`} stroke="#000000" strokeOpacity="0.35" strokeWidth="1" />
          {/* rebites */}
          {RIVETS.map((a, i) => (
            <circle key={i} cx={(39 * Math.cos(a)).toFixed(2)} cy={(39 * Math.sin(a)).toFixed(2)} r="1.5" fill="#ffffff" fillOpacity="0.5" />
          ))}
          {/* borda interna */}
          <circle r="35" fill="none" stroke={color} strokeOpacity="0.55" strokeWidth="1" />

          {/* centro abaulado */}
          <circle r="33" fill={`url(#dome-${uid})`} />
          {/* reflexo especular estático */}
          <ellipse cx="-10" cy="-14" rx="16" ry="9" fill="#ffffff" opacity="0.2" />

          {/* centro: ilustração 3D do catálogo, ou motivo geométrico (fallback) */}
          {art ? (
            <image
              href={art.src}
              x="-26"
              y="-26"
              width="52"
              height="52"
              clipPath={`url(#dome-clip-${uid})`}
              preserveAspectRatio="xMidYMid meet"
            />
          ) : (
            <g fill="#ffffff" fillOpacity="0.92" filter={`url(#emb-${uid})`}>
              <Motif kind={badge.kind} />
            </g>
          )}

          {/* faíscas */}
          {SPARKLES.map((sp, i) => {
            const rad = (sp.a * Math.PI) / 180
            const x = (sp.r * Math.cos(rad)).toFixed(2)
            const y = (sp.r * Math.sin(rad)).toFixed(2)
            return (
              <polygon
                key={i}
                points={sparkPoints(sp.s)}
                fill="#ffffff"
                fillOpacity="0.85"
                transform={`translate(${x},${y})`}
              />
            )
          })}

          {/* brilho varrendo o medalhão */}
          <g clipPath={`url(#dome-clip-${uid})`}>
            <g transform="rotate(-18)">
              <rect x="-9" y="-50" width="18" height="100" fill={`url(#shine-${uid})`}>
                <animateTransform
                  attributeName="transform"
                  type="translate"
                  values="-48 0; 60 0; 60 0"
                  keyTimes="0; 0.4; 1"
                  dur="3.6s"
                  repeatCount="indefinite"
                />
              </rect>
            </g>
          </g>
        </g>
      </svg>
    </span>
  )
}
```

interface BadgeArtPickerProps {

          type="button"
          aria-label={art.label}

          onClick={() => onChange(art.key)}
          className={`flex aspect-square items-center justify-center overflow-hidden rounded-lg border-2 bg-surface-container-highest p-1 transition-all ${
            value === art.key
              ? 'border-primary'
              : 'border-transparent hover:border-outline-variant/50'
          }`}
        >
          <img src={art.src} alt={art.label} className="h-full w-full object-contain" />
        </button>
      ))}
    </div>
  )
}
