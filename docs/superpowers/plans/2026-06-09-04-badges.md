# Legends — Plano 4: Selos (por limiar de votos) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Conceder selos automaticamente com base em votos recebidos — por categoria (ex.: 5 votos em Colaboração), por total acumulado (ex.: 25 votos) e por recorrência (reconhecido em N meses distintos) — sem qualquer noção de ranking, expondo o catálogo de selos e os selos conquistados por usuário.

**Architecture:** O model `Badge` ganha `threshold` (int) e `categorySlug` (opcional); `UserBadge` ganha unicidade `(userId, badgeId)` (selo concedido uma vez). Um `badge-service` calcula contagens de votos recebidos e concede os selos elegíveis de forma idempotente (`evaluateBadgesForUser`), além de listar os selos de um usuário (`listBadgesForUser`). A avaliação roda após cada voto bem-sucedido em `POST /votes`. Rotas novas: `GET /badges` (catálogo) e `GET /users/:id/badges` (conquistados). DTOs (`BadgeDTO`, `AwardedBadgeDTO`, enum `BadgeKind`) em `@legends/shared`. O frontend ganha uma página `/selos` mostrando o catálogo com os selos conquistados destacados.

**Tech Stack:** Fastify, Prisma (migration aditiva via `migrate diff` + `migrate deploy`), Vitest; React, React Router, TanStack Query, Testing Library.

**Prerequisites:** Planos 1–3 concluídos; ranking já removido (`BadgeKind = CATEGORY | RECURRENCE | IMPACT`, sem `MonthlyRankingEntry`). Postgres rodando. Banco de teste `legends_test`. Front consome `@legends/shared` só com `import type`.

```ts
import type { Badge, UserBadge } from "@prisma/client";
import { prisma } from "../lib/prisma";

function countVotesInCategory(
  userId: string,
  categorySlug: string,
): Promise<number> {
  return prisma.vote.count({
    where: { votedId: userId, category: { slug: categorySlug } },
  });
}

function countTotalVotes(userId: string): Promise<number> {
  return prisma.vote.count({ where: { votedId: userId } });
}

async function countDistinctMonths(userId: string): Promise<number> {
  const rows = await prisma.vote.findMany({
    where: { votedId: userId },
    distinct: ["periodId"],
    select: { periodId: true },
  });
  return rows.length;
}

async function qualifies(userId: string, badge: Badge): Promise<boolean> {
  if (badge.kind === "CATEGORY") {
    if (!badge.categorySlug) return false;
    return (
      (await countVotesInCategory(userId, badge.categorySlug)) >=
      badge.threshold
    );
  }
  if (badge.kind === "IMPACT") {
    return (await countTotalVotes(userId)) >= badge.threshold;
  }
  if (badge.kind === "RECURRENCE") {
    return (await countDistinctMonths(userId)) >= badge.threshold;
  }
  return false;
}

export async function evaluateBadgesForUser(
  userId: string,
): Promise<UserBadge[]> {
  const [badges, owned] = await Promise.all([
    prisma.badge.findMany(),
    prisma.userBadge.findMany({ where: { userId }, select: { badgeId: true } }),
  ]);
  const ownedIds = new Set(owned.map((entry) => entry.badgeId));

  const newlyAwarded: UserBadge[] = [];
  for (const badge of badges) {
    if (ownedIds.has(badge.id)) continue;
    if (await qualifies(userId, badge)) {
      newlyAwarded.push(
        await prisma.userBadge.create({ data: { userId, badgeId: badge.id } }),
      );
    }
  }
  return newlyAwarded;
}

export function listBadgesForUser(userId: string) {
  return prisma.userBadge.findMany({
    where: { userId },
    include: { badge: true },
    orderBy: { awardedAt: "desc" },
  });
}
```

```ts
import type { FastifyInstance } from "fastify";
import { z } from "zod";
import { MIN_JUSTIFICATION_LENGTH } from "@legends/shared";
import {
  VoteError,
  createVote,
  getCurrentOpenPeriod,
  listVotesByVoter,
} from "../services/voting-service";
import { evaluateBadgesForUser } from "../services/badge-service";
import { toVoteDTO } from "../lib/serialize";

const createVoteSchema = z.object({
  votedId: z.string().min(1),
  categoryId: z.string().min(1),
  justification: z.string().trim().min(MIN_JUSTIFICATION_LENGTH),
});

export async function voteRoutes(app: FastifyInstance) {
  app.post(
    "/votes",
    { onRequest: [app.authenticate] },
    async (request, reply) => {
      const parsed = createVoteSchema.safeParse(request.body);
      if (!parsed.success) {
        return reply
          .code(400)
          .send({
            message: "Justificativa e seleção obrigatórias.",
            issues: parsed.error.flatten(),
          });
      }
      try {
        const vote = await createVote({
          voterId: request.user.sub,
          votedId: parsed.data.votedId,
          categoryId: parsed.data.categoryId,
          justification: parsed.data.justification,
        });
        await evaluateBadgesForUser(vote.votedId);
        return reply.code(201).send({ vote: toVoteDTO(vote) });
      } catch (err) {
        if (err instanceof VoteError) {
          return reply.code(err.status).send({ message: err.message });
        }
        throw err;
      }
    },
  );

  app.get(
    "/votes/me",
    { onRequest: [app.authenticate] },
    async (request, reply) => {
      const period = await getCurrentOpenPeriod();
      if (!period) {
        return reply.send({ votes: [] });
      }
      const votes = await listVotesByVoter(request.user.sub, period.id);
      return reply.send({ votes: votes.map(toVoteDTO) });
    },
  );
}
```

```ts
export function buildApp(): FastifyInstance {
  const app = Fastify({ logger: false });

  app.register(cors, { origin: true });
  app.register(jwt, {
    secret: process.env.JWT_SECRET ?? "dev-secret-change-me",
  });

  app.decorate("authenticate", async function (request, reply) {
    try {
      await request.jwtVerify();
    } catch {
      return reply.code(401).send({ message: "Não autorizado" });
    }
  });

  app.get("/health", async () => ({ status: "ok" }));

  app.register(authRoutes, { prefix: "/auth" });
  app.register(userRoutes);
  app.register(categoryRoutes);
  app.register(periodRoutes);
  app.register(voteRoutes);
  app.register(badgeRoutes);

  return app;
}
```

```tsx
import { useQuery } from "@tanstack/react-query";
```
