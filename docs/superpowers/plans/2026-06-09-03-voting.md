# Legends — Plano 3: Votação Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Permitir que um usuário autenticado vote em um colega numa categoria com justificativa obrigatória, aplicando todas as regras do PRD §9, além de listar categorias, o período de votação corrente e os votos já registrados pelo próprio usuário no mês.

**Architecture:** O backend ganha um `voting-service` com `getCurrentOpenPeriod`, `createVote` (todas as regras) e `listVotesByVoter`, mais rotas `GET /categories`, `GET /periods/current`, `POST /votes` e `GET /votes/me`. DTOs (`CategoryDTO`, `VotingPeriodDTO`, `VoteDTO`, `CreateVoteRequest`) vivem em `@legends/shared`; a constante `MIN_JUSTIFICATION_LENGTH` também (usada pelo Zod no backend). O frontend ganha uma página de Votação (`/votar`) com React Query (queries de colegas/categorias/período/meus-votos + mutation de voto) e validação client-side espelhando o backend.

**Tech Stack:** Fastify, Prisma (constraint única `voterId+votedId+categoryId+periodId`), Zod, Vitest; React, React Router, TanStack Query, Testing Library.

**Prerequisites:** Planos 1 e 2 concluídos. Postgres rodando. Banco de teste `legends_test` configurado (truncate por teste). O front consome `@legends/shared` **apenas com `import type`**; valores runtime (constante de validação) são duplicados localmente no front.

```ts
import { Prisma, type VotingPeriod } from "@prisma/client";
import { prisma } from "../lib/prisma";

export class VoteError extends Error {
  constructor(
    message: string,
    public status: number,
  ) {
    super(message);
    this.name = "VoteError";
  }
}

export const voteInclude = {
  voter: true,
  voted: true,
  category: true,
  period: true,
} as const;

export type VoteWithRelations = Prisma.VoteGetPayload<{
  include: typeof voteInclude;
}>;

export function getCurrentOpenPeriod(): Promise<VotingPeriod | null> {
  return prisma.votingPeriod.findFirst({
    where: { status: "OPEN" },
    orderBy: { startsAt: "desc" },
  });
}

interface CreateVoteInput {
  voterId: string;
  votedId: string;
  categoryId: string;
  justification: string;
}

export async function createVote(
  input: CreateVoteInput,
): Promise<VoteWithRelations> {
  if (input.voterId === input.votedId) {
    throw new VoteError("Você não pode votar em si mesmo.", 400);
  }

  const period = await getCurrentOpenPeriod();
  if (!period) {
    throw new VoteError("Nenhum período de votação aberto no momento.", 409);
  }

  const voted = await prisma.user.findUnique({ where: { id: input.votedId } });
  if (!voted || !voted.active) {
    throw new VoteError("Colega inválido para votação.", 400);
  }

  const category = await prisma.category.findUnique({
    where: { id: input.categoryId },
  });
  if (!category || !category.active) {
    throw new VoteError("Categoria inválida.", 400);
  }

  try {
    return await prisma.vote.create({
      data: {
        voterId: input.voterId,
        votedId: input.votedId,
        categoryId: input.categoryId,
        periodId: period.id,
        justification: input.justification,
      },
      include: voteInclude,
    });
  } catch (err) {
    if (
      err instanceof Prisma.PrismaClientKnownRequestError &&
      err.code === "P2002"
    ) {
      throw new VoteError(
        "Você já votou nesse colega nessa categoria neste mês.",
        409,
      );
    }
    throw err;
  }
}

export function listVotesByVoter(
  voterId: string,
  periodId: string,
): Promise<VoteWithRelations[]> {
  return prisma.vote.findMany({
    where: { voterId, periodId },
    include: voteInclude,
    orderBy: { createdAt: "desc" },
  });
}
```

```ts
import type { FastifyInstance } from "fastify";
import { z } from "zod";
import { MIN_JUSTIFICATION_LENGTH } from "@legends/shared";
import {
  VoteError,
  createVote,
  getCurrentOpenPeriod,
  listVotesByVoter,
} from "../services/voting-service";
import { toVoteDTO } from "../lib/serialize";

const createVoteSchema = z.object({
  votedId: z.string().min(1),
  categoryId: z.string().min(1),
  justification: z.string().trim().min(MIN_JUSTIFICATION_LENGTH),
});

export async function voteRoutes(app: FastifyInstance) {
  app.post(
    "/votes",
    { onRequest: [app.authenticate] },
    async (request, reply) => {
      const parsed = createVoteSchema.safeParse(request.body);
      if (!parsed.success) {
        return reply
          .code(400)
          .send({
            message: "Justificativa e seleção obrigatórias.",
            issues: parsed.error.flatten(),
          });
      }
      try {
        const vote = await createVote({
          voterId: request.user.sub,
          votedId: parsed.data.votedId,
          categoryId: parsed.data.categoryId,
          justification: parsed.data.justification,
        });
        return reply.code(201).send({ vote: toVoteDTO(vote) });
      } catch (err) {
        if (err instanceof VoteError) {
          return reply.code(err.status).send({ message: err.message });
        }
        throw err;
      }
    },
  );

  app.get(
    "/votes/me",
    { onRequest: [app.authenticate] },
    async (request, reply) => {
      const period = await getCurrentOpenPeriod();
      if (!period) {
        return reply.send({ votes: [] });
      }
      const votes = await listVotesByVoter(request.user.sub, period.id);
      return reply.send({ votes: votes.map(toVoteDTO) });
    },
  );
}
```
