# Legends — Plano 6: Administração Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Permitir que administradores gerenciem categorias (criar, editar, ativar/desativar), abram e fechem o período de votação e moderem conteúdo (remover voto inadequado), com acesso restrito por papel (ADMIN).

**Architecture:** Um decorator `requireAdmin` (checa `request.user.role === 'ADMIN'` após `authenticate`) protege as rotas `/admin/*`. Um `admin-service` encapsula a lógica de período (`openVotingPeriod`/`closeVotingPeriod`/`monthRefFor`); um util `slugify` gera slugs de categoria a partir do nome. Rotas: `GET/POST /admin/categories`, `PATCH /admin/categories/:id`, `POST /admin/periods`, `POST /admin/periods/:id/close`, `DELETE /admin/votes/:id`. `CategoryDTO` ganha `active` (para a tela admin distinguir ativas/inativas). O frontend ganha `/admin` (só visível a admins) gerenciando categorias e o período.

**Tech Stack:** Fastify (decorator de autorização), Prisma, Zod, Vitest; React, React Router, TanStack Query, Testing Library.

**Prerequisites:** Planos 1–5 concluídos. Postgres rodando. Banco de teste `legends_test`. Há um usuário admin no seed (`admin@empresa.com`, papel ADMIN). Front consome `@legends/shared` só com `import type`.

export function slugify(value: string): string {
  return value
    .normalize("NFD")
    .replace(/[̀-ͯ]/g, "")
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}

import type { VotingPeriod } from "@prisma/client";
import { prisma } from "../lib/prisma";

export function monthRefFor(date: Date): string {
  return `${date.getFullYear()}-${String(date.getMonth() + 1).padStart(2, "0")}`;
}

export function openVotingPeriod(monthRef?: string): Promise<VotingPeriod> {
  const ref = monthRef ?? monthRefFor(new Date());
  const [year, month] = ref.split("-").map(Number);
  const startsAt = new Date(year, month - 1, 1);
  const endsAt = new Date(year, month, 0, 23, 59, 59);
  return prisma.votingPeriod.create({
    data: { monthRef: ref, startsAt, endsAt, status: "OPEN" },
  });
}

export function closeVotingPeriod(id: string): Promise<VotingPeriod> {
  return prisma.votingPeriod.update({
    where: { id },
    data: { status: "CLOSED" },
  });
}
```

import Fastify, { type FastifyInstance } from "fastify";
import cors from "@fastify/cors";
import jwt from "@fastify/jwt";
import { authRoutes } from "./routes/auth";
import { userRoutes } from "./routes/users";
import { categoryRoutes } from "./routes/categories";
import { periodRoutes } from "./routes/periods";
import { voteRoutes } from "./routes/votes";
import { badgeRoutes } from "./routes/badges";
import { profileRoutes } from "./routes/profile";
import { adminRoutes } from "./routes/admin";

export function buildApp(): FastifyInstance {
  const app = Fastify({ logger: false });

  app.register(cors, { origin: true });
  app.register(jwt, {
    secret: process.env.JWT_SECRET ?? "dev-secret-change-me",
  });

  app.decorate("authenticate", async function (request, reply) {
    try {
      await request.jwtVerify();
    } catch {
      return reply.code(401).send({ message: "Não autorizado" });
    }
  });

  app.decorate("requireAdmin", async function (request, reply) {
    if (request.user.role !== "ADMIN") {
      return reply
        .code(403)
        .send({ message: "Acesso restrito a administradores" });
    }
  });

  app.get("/health", async () => ({ status: "ok" }));

  app.register(authRoutes, { prefix: "/auth" });
  app.register(userRoutes);
  app.register(categoryRoutes);
  app.register(periodRoutes);
  app.register(voteRoutes);
  app.register(badgeRoutes);
  app.register(profileRoutes);
  app.register(adminRoutes);

  return app;
}

import type { FastifyInstance } from "fastify";
import { Prisma } from "@prisma/client";
import { z } from "zod";
import { prisma } from "../lib/prisma";
import { slugify } from "../lib/slug";
import { closeVotingPeriod, openVotingPeriod } from "../services/admin-service";
import { toCategoryDTO, toPeriodDTO } from "../lib/serialize";

const createCategorySchema = z.object({
  name: z.string().min(1),
  description: z.string().optional(),
});
const updateCategorySchema = z.object({
  name: z.string().min(1).optional(),
  description: z.string().optional(),
  active: z.boolean().optional(),
});
const openPeriodSchema = z.object({
  monthRef: z
    .string()
    .regex(/^\d{4}-\d{2}$/)
    .optional(),
});

export async function adminRoutes(app: FastifyInstance) {
  const adminOnly = { onRequest: [app.authenticate, app.requireAdmin] };

  app.get("/admin/categories", adminOnly, async (_request, reply) => {
    const categories = await prisma.category.findMany({
      orderBy: { name: "asc" },
    });
    return reply.send({ categories: categories.map(toCategoryDTO) });
  });

  app.post("/admin/categories", adminOnly, async (request, reply) => {
    const parsed = createCategorySchema.safeParse(request.body);
    if (!parsed.success) {
      return reply.code(400).send({ message: "Dados inválidos" });
    }
    try {
      const category = await prisma.category.create({
        data: {
          name: parsed.data.name,
          slug: slugify(parsed.data.name),
          description: parsed.data.description,
        },
      });
      return reply.code(201).send({ category: toCategoryDTO(category) });
    } catch (err) {
      if (
        err instanceof Prisma.PrismaClientKnownRequestError &&
        err.code === "P2002"
      ) {
        return reply
          .code(409)
          .send({ message: "Já existe uma categoria com esse nome." });
      }
      throw err;
    }
  });

  app.patch("/admin/categories/:id", adminOnly, async (request, reply) => {
    const { id } = request.params as { id: string };
    const parsed = updateCategorySchema.safeParse(request.body);
    if (!parsed.success) {
      return reply.code(400).send({ message: "Dados inválidos" });
    }
    try {
      const category = await prisma.category.update({
        where: { id },
        data: parsed.data,
      });
      return reply.send({ category: toCategoryDTO(category) });
    } catch (err) {
      if (
        err instanceof Prisma.PrismaClientKnownRequestError &&
        err.code === "P2025"
      ) {
        return reply.code(404).send({ message: "Categoria não encontrada" });
      }
      throw err;
    }
  });

  app.post("/admin/periods", adminOnly, async (request, reply) => {
    const parsed = openPeriodSchema.safeParse(request.body ?? {});
    if (!parsed.success) {
      return reply
        .code(400)
        .send({ message: "monthRef inválido (use YYYY-MM)" });
    }
    try {
      const period = await openVotingPeriod(parsed.data.monthRef);
      return reply.code(201).send({ period: toPeriodDTO(period) });
    } catch (err) {
      if (
        err instanceof Prisma.PrismaClientKnownRequestError &&
        err.code === "P2002"
      ) {
        return reply
          .code(409)
          .send({ message: "Já existe um período para esse mês." });
      }
      throw err;
    }
  });

  app.post("/admin/periods/:id/close", adminOnly, async (request, reply) => {
    const { id } = request.params as { id: string };
    try {
      const period = await closeVotingPeriod(id);
      return reply.send({ period: toPeriodDTO(period) });
    } catch (err) {
      if (
        err instanceof Prisma.PrismaClientKnownRequestError &&
        err.code === "P2025"
      ) {
        return reply.code(404).send({ message: "Período não encontrado" });
      }
      throw err;
    }
  });

  app.delete("/admin/votes/:id", adminOnly, async (request, reply) => {
    const { id } = request.params as { id: string };
    try {
      await prisma.vote.delete({ where: { id } });
      return reply.code(204).send();
    } catch (err) {
      if (
        err instanceof Prisma.PrismaClientKnownRequestError &&
        err.code === "P2025"
      ) {
        return reply.code(404).send({ message: "Voto não encontrado" });
      }
      throw err;
    }
  });
}
```

import { useState, type FormEvent } from "react";
import { Link } from "react-router-dom";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import type { CategoryDTO, VotingPeriodDTO } from "@legends/shared";
import { ApiError, apiFetch } from "../lib/api";
import { useAuth } from "../auth/AuthContext";

export function AdminPage() {
  const { user } = useAuth();
  const queryClient = useQueryClient();
  const [name, setName] = useState("");
  const [error, setError] = useState<string | null>(null);

  const isAdmin = user?.role === "ADMIN";

  const categoriesQuery = useQuery({
    queryKey: ["admin", "categories"],
    queryFn: () => apiFetch<{ categories: CategoryDTO[] }>("/admin/categories"),
    enabled: isAdmin,
  });
  const periodQuery = useQuery({
    queryKey: ["period"],
    queryFn: () =>
      apiFetch<{ period: VotingPeriodDTO | null }>("/periods/current"),
    enabled: isAdmin,
  });

  const createCategory = useMutation({
    mutationFn: (body: { name: string }) =>
      apiFetch<{ category: CategoryDTO }>("/admin/categories", {
        method: "POST",
        body: JSON.stringify(body),
      }),
    onSuccess: () => {
      setName("");
      setError(null);
      queryClient.invalidateQueries({ queryKey: ["admin", "categories"] });
    },
    onError: (err) =>
      setError(
        err instanceof ApiError ? err.message : "Erro ao criar categoria.",
      ),
  });
  const toggleCategory = useMutation({
    mutationFn: (vars: { id: string; active: boolean }) =>
      apiFetch<{ category: CategoryDTO }>(`/admin/categories/${vars.id}`, {
        method: "PATCH",
        body: JSON.stringify({ active: vars.active }),
      }),
    onSuccess: () =>
      queryClient.invalidateQueries({ queryKey: ["admin", "categories"] }),
  });
  const openPeriod = useMutation({
    mutationFn: () =>
      apiFetch<{ period: VotingPeriodDTO }>("/admin/periods", {
        method: "POST",
        body: JSON.stringify({}),
      }),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ["period"] }),
    onError: (err) =>
      setError(
        err instanceof ApiError ? err.message : "Erro ao abrir período.",
      ),
  });
  const closePeriod = useMutation({
    mutationFn: (id: string) =>
      apiFetch<{ period: VotingPeriodDTO }>(`/admin/periods/${id}/close`, {
        method: "POST",
      }),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ["period"] }),
  });

  if (!isAdmin) {
    return (
      <p className="p-8 text-red-400">Acesso restrito a administradores.</p>
    );
  }

  function handleCreate(event: FormEvent) {
    event.preventDefault();
    if (!name.trim()) return;
    createCategory.mutate({ name: name.trim() });
  }

  const period = periodQuery.data?.period ?? null;

  return (
    <main className="min-h-screen p-8 flex flex-col gap-8">
      <header className="flex items-center justify-between">
        <h1 className="text-3xl font-extrabold tracking-tight">
          Administração
        </h1>
        <Link to="/" className="text-sm text-slate-400 hover:text-slate-100">
          ← Voltar
        </Link>
      </header>

      <section className="flex flex-col gap-3">
        <h2 className="font-mono text-xs uppercase tracking-widest text-secondary">
          Período de votação
        </h2>
        {period ? (
          <div className="bg-surface rounded-lg p-4 flex items-center justify-between">
            <span>
              Aberto:{" "}
              <span className="font-mono text-secondary">
                {period.monthRef}
              </span>
            </span>
            <button
              onClick={() => closePeriod.mutate(period.id)}
              className="text-sm rounded border border-slate-600 px-3 py-1 hover:border-primary"
            >
              Fechar período
            </button>
          </div>
        ) : (
          <div className="bg-surface rounded-lg p-4 flex items-center justify-between">
            <span className="text-slate-300">Nenhum período aberto.</span>
            <button
              onClick={() => openPeriod.mutate()}
              className="text-sm rounded bg-primary text-primary-fg px-3 py-1 font-semibold hover:opacity-90"
            >
              Abrir período do mês
            </button>
          </div>
        )}
      </section>

      <section className="flex flex-col gap-3">
        <h2 className="font-mono text-xs uppercase tracking-widest text-secondary">
          Categorias
        </h2>
        <form onSubmit={handleCreate} className="flex gap-2">
          <input
            value={name}
            onChange={(event) => setName(event.target.value)}
            aria-label="Nova categoria"
            placeholder="Nova categoria"
            className="flex-1 rounded bg-bg border border-slate-700 px-3 py-2 text-slate-100 focus:border-primary focus:outline-none"
          />
          <button
            type="submit"
            className="rounded bg-primary text-primary-fg px-4 py-2 font-semibold hover:opacity-90"
          >
            Adicionar
          </button>
        </form>
        {error && (
          <p role="alert" className="text-sm text-red-400">
            {error}
          </p>
        )}
        <ul className="flex flex-col gap-2">
          {categoriesQuery.data?.categories.map((category) => (
            <li
              key={category.id}
              className="bg-surface rounded-lg p-3 flex items-center justify-between"
            >
              <span
                className={category.active ? "" : "text-slate-500 line-through"}
              >
                {category.name}
              </span>
              <button
                onClick={() =>
                  toggleCategory.mutate({
                    id: category.id,
                    active: !category.active,
                  })
                }
                className="text-xs font-mono rounded border border-slate-600 px-2 py-1 hover:border-primary"
              >
                {category.active ? "Desativar" : "Ativar"}
              </button>
            </li>
          ))}
        </ul>
      </section>
    </main>
  );
}
